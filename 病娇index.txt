let currentQ = 0;
let scoreArr = Array(questions.length).fill(null);

// 渲染题目
function renderQuestion(){
  const wrap = document.getElementById("question-wrap");
  wrap.innerHTML = "";
  const q = questions[currentQ];
  let html = `<div class="page active" id="qpage-${currentQ}">
    <div class="q-index">第 ${currentQ+1} / ${questions.length} 题</div>
    <div class="q-title">${q.q}</div>`;
  q.o.forEach((opt,idx)=>{
    const selClass = scoreArr[currentQ] === idx ? "selected" : "";
    html += `<div class="option-item ${selClass}" onclick="selectOpt(${idx})">${opt.t}</div>`;
  })
  html += `<div class="next-wrap"><button class="btn" onclick="nextQuestion()">下一题</button></div></div>`;
  wrap.innerHTML = html;
}

function selectOpt(idx){
  scoreArr[currentQ] = idx;
  renderQuestion();
}

function nextQuestion(){
  if(scoreArr[currentQ] === null){
    alert("请选择一个选项");
    return;
  }
  currentQ++;
  if(currentQ >= questions.length){
    calcResult();
    goTo("result");
  }else{
    renderQuestion();
  }
}

function calcResult(){
  let total = 0;
  for(let i=0;i<questions.length;i++){
    const sIdx = scoreArr[i];
    total += questions[i].o[sIdx].s;
  }
  let res = resultList[0];
  for(let r of resultList){
    if(total >= r.min && total <= r.max){
      res = r;
      break;
    }
  }
  document.getElementById("res-title").innerText = res.title;
  document.getElementById("res-content").innerText = res.txt;
}

function goTo(target){
  document.querySelectorAll(".page").forEach(p=>p.classList.remove("active"));
  if(target === 1){
    currentQ = 0;
    renderQuestion();
  }else if(target === "result"){
    document.getElementById("page-result").classList.add("active");
  }
}

function restartTest(){
  currentQ = 0;
  scoreArr = Array(questions.length).fill(null);
  goTo(1);
}
